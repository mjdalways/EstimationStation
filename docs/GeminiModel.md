1. The JQL (Jira Query Language)
The "magic" happens in the JQL query. To find stories in a specific sprint that have no points and aren't bugs or spikes, your query will look like this:

project = "YOUR_PROJ" AND sprint = {sprintId} AND issuetype = "Story" AND "Story Points" is EMPTY

Note: If you want all types except bugs/spikes, use:
sprint = {sprintId} AND issuetype NOT IN ("Bug", "Spike") AND "Story Points" is EMPTY

2. Setting Up Authentication
You shouldn't use your password. Instead:

Go to Atlassian Account Settings > Security > API Tokens.

Create a token and save it.

In C#, you will use Basic Auth, which is a Base64 encoded string of your-email@example.com:your-api-token.

Here is a concise way to fetch these issues using System.Net.Http.

using System.Net.Http.Headers;
using System.Text;
using System.Text.Json;

public async Task<List<JiraIssue>> GetUnpointedStories(string sprintId)
{
    var email = "your-email@example.com";
    var apiToken = "your-api-token";
    var domain = "your-domain.atlassian.net";
    
    // 1. Encode Credentials
    var authHeader = Convert.ToBase64String(Encoding.ASCII.GetBytes($"{email}:{apiToken}"));

    using var client = new HttpClient();
    client.BaseAddress = new Uri($"https://{domain}/");
    client.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Basic", authHeader);

    // 2. Build JQL
    var jql = $"sprint = {sprintId} AND issuetype = 'Story' AND 'Story Points' is EMPTY";
    var url = $"/rest/api/3/search?jql={Uri.EscapeDataString(jql)}&fields=summary,description,issuetype";

    // 3. Execute
    var response = await client.GetAsync(url);
    if (response.IsSuccessStatusCode)
    {
        var json = await response.Content.ReadAsStringAsync();
        // Parse the 'issues' array from the JSON response
        var result = JsonSerializer.Deserialize<JiraResponse>(json);
        return result.Issues;
    }
    
    return new List<JiraIssue>();
}

public class JiraIssueService

    private readonly HttpClient _httpClient;
    private readonly string _jiraDomain;
    private readonly string _authHeaderValue;

    public JiraIssueService(HttpClient httpClient, string email, string apiToken, string jiraDomain)
    {
        _httpClient = httpClient ?? throw new ArgumentNullException(nameof(httpClient));
        _jiraDomain = jiraDomain ?? throw new ArgumentNullException(nameof(jiraDomain));
        
        // Construct Basic Authentication string (Base64 encoded email:token)
        var credentials = $"{email}:{apiToken}";
        var bytes = Encoding.ASCII.GetBytes(credentials);
        _authHeaderValue = Convert.ToBase64String(bytes);
    }

    public async Task<List<JiraIssueDto>> GetUnpointedStoriesAsync(string sprintId)
    {
        _httpClient.BaseAddress = new Uri($"https://{_jiraDomain}/");
        _httpClient.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Basic", _authHeaderValue);
        _httpClient.DefaultRequestHeaders.Accept.Add(new MediaTypeWithQualityHeaderValue("application/json"));

        // JQL Filter: Target specific sprint, ensure type is Story, verify Story Points are empty
        // NOTE: If your instance requires a custom field ID, substitute 'Story Points' with 'customfield_XXXXX'
        string jql = $"sprint = {sprintId} AND issuetype = 'Story' AND 'Story Points' is EMPTY";
        string fields = "summary,description,issuetype,status";
        string requestUrl = $"/rest/api/3/search?jql={Uri.EscapeDataString(jql)}&fields={Uri.EscapeDataString(fields)}";

        HttpResponseMessage response = await _httpClient.GetAsync(requestUrl);
        
        if (!response.IsSuccessStatusCode)

        return result?.Issues ?? new List<JiraIssueDto>();
    }
