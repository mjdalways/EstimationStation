Post-Reveal Dispatch Logic (room.js)

connection.on('VotesRevealed', (votes, stats) => {
  const beh    = getPostRevealBehavior();
  const groups = buildVoteGroups(roomState.participants, votes);

  // Boss fight visual wraps the whole reveal if the ticket qualifies
  if (beh.bossFightEnabled && currentCardIsBoss(beh.bossFightMinPoints)) {
    enterBossArena(stats);
  }

  if (stats.isConsensus) {
    if (beh.celebrationEnabled) triggerCelebration(stats);
    if (beh.streaksEnabled)     incrementStreak(stats);
    if (beh.finisherEnabled && isHardWonConsensus(stats)) triggerFinisher();
    return;
  } else if (beh.streaksEnabled) {
    resetStreak();
  }

  const hasOutlier = !!stats.shameParticipantId;
  const hasBattle  = groups.length >= 2
                  && totalVoters(groups) >= beh.minVotersForBattle;

  if (hasOutlier && beh.shameEnabled
      && (beh.outlierBehavior === 'shame-only' || beh.outlierBehavior === 'both')) {
    triggerShame(stats.shameParticipantName, stats.shameParticipantId,
                  stats.majorityValue, stats.outlierValue);
  }

  const shameBlocksBattle = hasOutlier && beh.outlierBehavior === 'shame-only';
  if (hasBattle && beh.battleEnabled && !shameBlocksBattle) {
    if (beh.duelEnabled && isStraightOneVOne(groups)) {
      triggerDuelOfPerspective(groups);
    } else {
      triggerBattle(groups);
    }
  }

  if (beh.discussionTimerMode !== 'off') {
    startDiscussionTimer(beh.discussionTimerMode, beh.discussionThresholdMin);
  }
});

4. Top-Priority — Shame (a.k.a. The Lone Wolf)

Trigger Condition After votes are revealed: exactly one participant voted a different numeric value from all others (n−1 voters agree on value X; 1 voter chose value Y). Non-numeric votes (?, ☕) are excluded. The outlier is called out with animation and emoji, then the overlay transitions to a positive message encouraging them to make their case.
Naming note from Gemini brainstorm Gemini proposed renaming this from "Shame" to "The Lone Wolf" or "The Holdout" with a spotlight + cricket effect for lighthearted framing. The plan keeps Shame as the internal mechanic name (existing copy) but exposes display label "Lone Wolf" in the settings UI and adds a cricket SFX + tumbleweed option in §4.6.

4.2 Backend — PokerHub.cs — Outlier Detection

Appended to CalculateStats() after the existing consensus check:

// Single-outlier detection
if (!stats.IsConsensus && numericVotes.Count >= 2)
{
    var groups = numericVotes
        .GroupBy(kvp => kvp.Value)
        .OrderByDescending(g => g.Count())
        .ToList();

    if (groups.Count == 2 && groups[1].Count() == 1)
    {
        var outlierId = groups[1].First().Key;   // ConnectionId
        var outlier   = room.Participants
            .FirstOrDefault(p => p.ConnectionId == outlierId);
        if (outlier != null)
        {
            stats.ShameParticipantName = outlier.Name;
            stats.ShameParticipantId   = outlierId;
            stats.MajorityValue        = groups[0].Key;
            stats.OutlierValue         = groups[1].Key;
        }
    }
}

  cricketSfx:      false,

  displayLabel:    'Lone Wolf'

function getShameSettings() {
  try {
    const raw = localStorage.getItem('es_shameSettings');
    return { ...DEFAULT_SHAME, ...(raw ? JSON.parse(raw) : {}) };
  } catch { return { ...DEFAULT_SHAME }; }
}

function triggerShame(name, participantId, majorityValue, outlierValue) {
  const s = getShameSettings();
  const badge = document.querySelector(`[data-participant-id="${participantId}"]`);

  badge?.style.setProperty('--shame-color', s.shameColor);
  badge?.classList.add('shame-highlight');
  badge?.dataset.style = s.animationStyle;
  if (s.animationStyle === 'spotlight') addBackdrop();
  if (s.tumbleweed) spawnTumbleweed();
  if (s.cricketSfx) new Audio('/sfx/cricket.mp3').play().catch(()=>{});

  if (s.floatingEmojis) {
    const rect = badge?.getBoundingClientRect();
    for (let i = 0; i < 6; i++)
      setTimeout(() => spawnBooEmoji(s, rect), i * 220);
  }
  showShameToast(name, majorityValue, outlierValue, s);
  if (isMe(participantId) && canCounterSpell()) showCounterSpellButton(badge);
  setTimeout(() => cleanupShame(badge), s.duration);
}

5.3 Backend — PokerHub.cs — UpdateAvatar

public async Task UpdateAvatar(string avatarData)
{
    if (avatarData?.StartsWith("data:") == true && avatarData.Length > 65_536)
    {
        await Clients.Caller.SendAsync("Error", "Avatar image too large (max ~48 KB).");
        return;
    }
    var roomName    = _roomService.GetRoomForConnection(Context.ConnectionId);
    var room        = _roomService.GetRoom(roomName);
    var participant = room?.Participants
        .FirstOrDefault(p => p.ConnectionId == Context.ConnectionId);
    if (participant == null) return;
    participant.AvatarData = avatarData;
    await Clients.Group(roomName).SendAsync("AvatarUpdated", Context.ConnectionId, avatarData);
}

6.7 Phase Orchestrator (battle.js)

async function triggerBattle(groups) {
  const s   = getBattleSettings();
  const mul = { slow: 1.5, normal: 1, fast: 0.65 }[s.speed] ?? 1;
  const arena = buildArena(groups, s);
  document.body.appendChild(arena);

  await sleep(1500 * mul);  runLineup(arena);
  await sleep(1500 * mul);  runTaunt(arena, groups);

  const clashes = clashCountFor(groups);   // 2–4 based on team count
  for (let i = 0; i < clashes; i++) {
    await runOneClash(arena, s, i);
  }

  await sleep(300 * mul); runAftermath(arena, s);
  await sleep(2500 * mul); runResolution(arena, s);
  if (s.autoSkipMs > 0) setTimeout(() => dismissBattle(), s.autoSkipMs);
}

function enableVoiceReveal(phrase = 'reveal') {
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!SR) { showToast('Voice reveal not supported in this browser'); return; }
  const rec = new SR();
  rec.continuous     = true;
  rec.interimResults = false;
  rec.lang           = 'en-US';
  rec.onresult = (e) => {
    const t = e.results[e.results.length-1][0].transcript.toLowerCase().trim();
    if (t.includes(phrase.toLowerCase())) triggerRevealAsHost();
  };
  rec.start();
}

Caveat: Only the host needs to enable it. Display a small 🎤 indicator while listening so users know their mic is open.

7.3 C# Service Implementation

public class JiraIssueService
{
    private readonly HttpClient _httpClient;
    private readonly string _jiraDomain;
    private readonly string _authHeaderValue;
    private readonly string _storyPointsField;

    public JiraIssueService(HttpClient httpClient, string email, string apiToken,
                              string jiraDomain, string storyPointsField = "Story Points")
    {
        _httpClient       = httpClient;
        _jiraDomain       = jiraDomain;
        _storyPointsField = storyPointsField;
        var credentials  = $"{email}:{apiToken}";
        _authHeaderValue = Convert.ToBase64String(Encoding.ASCII.GetBytes(credentials));
    }

    public async Task<List<JiraIssueDto>> GetUnpointedStoriesAsync(string sprintId)
    {
        _httpClient.BaseAddress = new Uri($"https://{_jiraDomain}/");
        _httpClient.DefaultRequestHeaders.Authorization =
            new AuthenticationHeaderValue("Basic", _authHeaderValue);
        _httpClient.DefaultRequestHeaders.Accept.Add(
            new MediaTypeWithQualityHeaderValue("application/json"));

        var jql = $"sprint = {sprintId} AND issuetype = 'Story' AND '{_storyPointsField}' is EMPTY";
        var fields = "summary,description,issuetype,status";
        var url    = $"/rest/api/3/search?jql={Uri.EscapeDataString(jql)}&fields={Uri.EscapeDataString(fields)}";

        var response = await _httpClient.GetAsync(url);
        if (!response.IsSuccessStatusCode)
        {
            var err = await response.Content.ReadAsStringAsync();
            throw new HttpRequestException(
                $"Jira API call failed ({response.StatusCode}): {err}");
        }

        var json   = await response.Content.ReadAsStringAsync();
        var result = JsonSerializer.Deserialize<JiraSearchResponseDto>(json);
        return result?.Issues ?? new List<JiraIssueDto>();
    }
}

public class JiraSearchResponseDto {
    [JsonPropertyName("issues")] public List<JiraIssueDto> Issues { get; set; } = new();
}
public class JiraIssueDto {
    [JsonPropertyName("id")]     public string Id { get; set; } = "";
    [JsonPropertyName("key")]    public string Key { get; set; } = "";
    [JsonPropertyName("fields")] public IssueFields Fields { get; set; } = new();
}
public class IssueFields {
    [JsonPropertyName("summary")]     public string Summary { get; set; } = "";
    [JsonPropertyName("description")] public object? Description { get; set; }  // ADF object
    [JsonPropertyName("issuetype")]   public IssueType IssueType { get; set; } = new();
}
public class IssueType {
    [JsonPropertyName("name")] public string Name { get; set; } = "";
}

7.4 Custom-Field Caveat

Story Points field Many Jira Cloud instances expose Story Points only via a custom field ID (customfield_10016 or similar). The service constructor accepts an override. To discover it, the user can open any pointed ticket, append .json to the URL, and Ctrl+F for the value — adjacent key is the custom field ID.

    foreach (var issue in issues)
    {
        await Task.Delay(120);  // stagger for portal animation
        await Clients.Group(roomName).SendAsync("JiraIssueArrived", issue);
    }
    await Clients.Group(roomName).SendAsync("JiraImportCompleted");
}

7.6 Portal Import Animation

  1. Portal opens: Jira-blue swirling vortex (canvas-based, 1.2s ease-out scale-in) appears center-stage
  2. Vacuum effect: Each JiraIssueArrived event spawns a card flying out of the vortex on a parabolic path to its slot in the deck
  3. Dust-off: Imported cards have a sepia + 0.4 brightness CSS filter ("dusty"); class is removed when first vote is locked, accompanied by a 200ms CSS shake to symbolize "awakening"
  4. Portal closes: JiraImportCompleted triggers reverse animation; vortex collapses to a point and fades
